cmake_minimum_required(VERSION 3.20.0)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(pcmplay)
target_sources(app PRIVATE src/main.c)
# Plays every clips/*.pcm (raw 16 kHz s16le mono) in name order. clips/ is git-ignored (audio derived from TTS
# voices); tools/relp_reference.py "<text>" c/pcmplay/clips makes one set.
set(gen ${ZEPHYR_BINARY_DIR}/include/generated)
file(GLOB clips CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/clips/*.pcm)
list(SORT clips)
set(decls "")
set(table "")
set(i 0)
foreach(clip ${clips})
  get_filename_component(name ${clip} NAME_WLE)
  generate_inc_file_for_target(app ${clip} ${gen}/clip${i}.inc)
  string(APPEND decls "static const uint8_t clip${i}[] __aligned(2) = {\n#include <clip${i}.inc>\n};\n")
  string(APPEND table "  {\"${name}\", clip${i}, sizeof clip${i}},\n")
  math(EXPR i "${i} + 1")
endforeach()
file(WRITE ${gen}/clips.h "${decls}static const struct clip clips[] = {\n${table}};\n")
