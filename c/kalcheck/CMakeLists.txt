cmake_minimum_required(VERSION 3.20.0)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(kalcheck)
target_sources(app PRIVATE src/main.c ../kalsynth/kal_synth.c ../relpsynth/relp_synth.c)
target_include_directories(app PRIVATE ../kalsynth ../relpsynth ../lpcsynth)
# Plays every clips/*.wire (phones in tools/kal_diphones.py's wire format) in name order with the diphone table
# clips/kal-relp.bin, kept in flash. clips/ is git-ignored (CMU-kal-derived data; its notice is kal-relp.COPYING,
# which build writes beside the table and which goes wherever the table goes). Make them with:
#   tools/kal_diphones.py build c/kalcheck/clips/kal-relp.bin          (table + kal-relp.COPYING)
#   tools/kal_diphones.py flite "<text>" <dir> c/kalcheck/clips/kal-relp.bin && cp <dir>/phones.wire c/kalcheck/clips/<n>-<name>.wire
set(gen ${ZEPHYR_BINARY_DIR}/include/generated)
set(table ${CMAKE_CURRENT_SOURCE_DIR}/clips/kal-relp.bin)
if(NOT EXISTS ${table})
  message(FATAL_ERROR "no ${table}: see the comment in CMakeLists.txt")
endif()
generate_inc_file_for_target(app ${table} ${gen}/kal_table.inc)
file(GLOB phrases CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/clips/*.wire)
list(SORT phrases)
set(decls "static const uint8_t kal_table[] = {\n#include <kal_table.inc>\n};\n")
set(list "")
set(i 0)
foreach(phrase ${phrases})
  get_filename_component(name ${phrase} NAME_WLE)
  generate_inc_file_for_target(app ${phrase} ${gen}/phrase${i}.inc)
  string(APPEND decls "static const uint8_t phrase${i}[] = {\n#include <phrase${i}.inc>\n};\n")
  string(APPEND list "  {\"${name}\", phrase${i}, sizeof phrase${i}},\n")
  math(EXPR i "${i} + 1")
endforeach()
file(WRITE ${gen}/phrases.h "${decls}static const struct phrase phrases[] = {\n${list}};\n")
